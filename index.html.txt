function TierBadge({ color, active, size = 'md', progress = 0 }: { color: string, active?: boolean, size?: 'sm' | 'md' | 'lg', progress?: number }) {
  const badgeSize = size === 'sm' ? 'w-10 h-10' : size === 'md' ? 'w-14 h-14' : 'w-16 h-16';
  const innerSize = size === 'sm' ? 'w-7 h-7' : size === 'md' ? 'w-10 h-10' : 'w-12 h-12';
  const logoSize = size === 'sm' ? 'scale-[0.4]' : size === 'md' ? 'scale-[0.6]' : 'scale-[0.8]';

  return (
    <motion.div 
      animate={active ? {
        y: [0, -4, 0],
      } : {}}
      transition={{
        duration: 3,
        repeat: Infinity,
        ease: "easeInOut"
      }}
      className={`${badgeSize} rounded-full flex items-center justify-center relative`}
    >
      {/* Outer Glow for Active */}
      {active && (
        <motion.div 
          animate={{ opacity: [0.3, 0.6, 0.3], scale: [1, 1.1, 1] }}
          transition={{ duration: 2, repeat: Infinity }}
          className={`absolute inset-0 rounded-full blur-md ${color.replace('text-', 'bg-')}/30`}
        />
      )}
      
      {/* Textured Border (from image) */}
      <div className={`absolute inset-0 rounded-full border-[3px] border-white/20 p-[1px]`}>
        <div className="w-full h-full rounded-full border border-white/10 border-dashed opacity-50" />
      </div>
      
      {/* Central Circle */}
      <div className={`${innerSize} rounded-full bg-gradient-to-br from-white/30 via-transparent to-black/40 flex items-center justify-center shadow-2xl relative overflow-hidden`}>
        {/* Shimmer Effect */}
        <motion.div 
          animate={{ x: ['-100%', '200%'] }}
          transition={{ duration: 2, repeat: Infinity, ease: "linear", repeatDelay: 1 }}
          className="absolute inset-0 bg-gradient-to-r from-transparent via-white/30 to-transparent -skew-x-12"
        />
        <div className={`absolute inset-0 rounded-full opacity-60 ${color.replace('text-', 'bg-')}`} />
        
        {/* ABB Logo Shapes (from image) */}
        <div className={`relative z-10 flex flex-col items-center justify-center ${logoSize}`}>
          <div className="flex gap-1 items-end">
            <div className="w-3 h-6 bg-white rounded-full rotate-[-15deg] opacity-90" />
            <div className="w-3 h-8 bg-white rounded-full rotate-[-15deg] opacity-100" />
            <div className="w-3 h-5 bg-white rounded-full rotate-[-15deg] opacity-80" />
          </div>
        </div>
      </div>

      {/* Completed Checkmark (from image) */}
      {progress >= 100 && (
        <div className="absolute -bottom-1 -right-1 w-6 h-6 bg-green-500 rounded-full border-2 border-white flex items-center justify-center shadow-lg z-20">
          <Check size={12} className="text-white stroke-[4]" />

      )}

  );
}
